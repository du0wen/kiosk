# Kiosk P1 · Plan 4 — CLI (`@kiosk/cli`) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the `kiosk` CLI — the skill's primary path — so an AI tool can `kiosk publish <file>` and get back a public URL, with re-publish hitting the same URL.

**Architecture:** A standalone zero-dependency package in `cli/` (Node built-ins only: `util.parseArgs`, `fs`, `os`, `path`, global `fetch`). Four focused modules — `config` (token/apiBase resolution), `state` (`.kiosk/pages.json` + create-vs-update identity), `client` (REST calls), `commands`+`cli` (arg parsing + dispatch). All logic is unit/integration-tested with a temp filesystem and a mocked `fetch`; no running server needed.

**Tech Stack:** Node 20+ (ESM, `node:util` parseArgs, global fetch), TypeScript, Vitest (root runner extended to `cli/`).

> Plan **4 of 6**. Consumes the Plan 2 REST API shape (`POST/PUT/GET/DELETE /pages`, `GET /me`, Bearer auth, `{ error: { code, message } }`). Independent of the app's `src/` code. `install-skill` → Plan 5.

## Global Constraints

From `docs/specs/2026-07-08-kiosk-p1-publish-core-design.md` §2.2 and CLAUDE `cli.md`.

- Commands: `login`, `publish <file>`, `list`, `open <id|file>`, `delete <id|file>`, `whoami`, `logout`. (`install-skill` = Plan 5.)
- `publish` identity order: `--id <id>` → `--new` → `.kiosk/pages.json` lookup (by file path relative to the nearest `.kiosk/`) → else create + write mapping.
- HTML sent as a **file body** (`text/html`); title via `X-Kiosk-Title`. Default output = **the URL, one line**; `--json` for structured.
- Token: `KIOSK_TOKEN` env **takes precedence** over `~/.config/kiosk/config.json`; `apiBase` overridable via `KIOSK_API_BASE`.
- Errors → friendly message on stderr + **non-zero exit**; 401 → exit code 2.
- `.kiosk/pages.json` shape: `{ "version": 1, "pages": { "<relpath>": { "id", "url" } } }`. Do not auto-edit `.gitignore`.
- Commits carry **no AI attribution**.
- **Deferred:** `install-skill` + SKILL.md (Plan 5); real CLI→API→serving e2e is a manual smoke (documented, needs a running server + token).

import { readFileSync, writeFileSync, mkdirSync } from "node:fs";
import { homedir } from "node:os";
import { join, dirname } from "node:path";

const DEFAULT_API_BASE = "http://localhost:3000/api/v1";

export function configPath(): string {
  return join(homedir(), ".config", "kiosk", "config.json");
}

export type Config = { token?: string; apiBase: string };

export function resolveConfig(env: Record<string, string | undefined> = process.env): Config {
  let fileCfg: Partial<Config> = {};
  try {
    fileCfg = JSON.parse(readFileSync(configPath(), "utf8"));
  } catch {
    // no config file yet
  }
  return {
    token: env.KIOSK_TOKEN || fileCfg.token,
    apiBase: env.KIOSK_API_BASE || fileCfg.apiBase || DEFAULT_API_BASE,
  };
}

export function saveConfig(cfg: Config): void {
  const p = configPath();
  mkdirSync(dirname(p), { recursive: true });
  writeFileSync(p, JSON.stringify(cfg, null, 2), { mode: 0o600 });
}
```

import { readFileSync, writeFileSync, mkdirSync, existsSync } from "node:fs";
import { join, dirname, relative, resolve } from "node:path";

const STATE_DIR = ".kiosk";
const STATE_FILE = "pages.json";

type State = {
  version: number;
  pages: Record<string, { id: string; url: string }>;
};

function statePath(root: string): string {
  return join(root, STATE_DIR, STATE_FILE);
}

/** Nearest ancestor of startDir containing a .kiosk/ dir; else startDir itself. */
export function findProjectRoot(startDir: string): string {
  let dir = resolve(startDir);
  for (;;) {
    if (existsSync(join(dir, STATE_DIR))) return dir;
    const parent = dirname(dir);
    if (parent === dir) return resolve(startDir);
    dir = parent;
  }
}

export function mappingKey(root: string, file: string): string {
  return relative(root, resolve(file));
}

export function readState(root: string): State {
  try {
    return JSON.parse(readFileSync(statePath(root), "utf8")) as State;
  } catch {
    return { version: 1, pages: {} };
  }
}

export function writeMapping(
  root: string,
  key: string,
  entry: { id: string; url: string },
): void {
  const state = readState(root);
  state.pages[key] = entry;
  const p = statePath(root);
  mkdirSync(dirname(p), { recursive: true });
  writeFileSync(p, JSON.stringify(state, null, 2));
}

export type Identity = { mode: "update"; id: string } | { mode: "create" };

export function resolveIdentity(opts: {
  root: string;
  file: string;
  id?: string;
  forceNew?: boolean;
}): Identity {
  if (opts.id) return { mode: "update", id: opts.id };
  if (opts.forceNew) return { mode: "create" };
  const existing = readState(opts.root).pages[mappingKey(opts.root, opts.file)];
  return existing ? { mode: "update", id: existing.id } : { mode: "create" };
}
```

export class CliError extends Error {
  constructor(
    message: string,
    public exitCode = 1,
  ) {
    super(message);
    this.name = "CliError";
  }
}

export type PageResult = { id: string; url: string; [k: string]: unknown };

export class KioskClient {
  constructor(
    private apiBase: string,
    private token: string,
  ) {}

  private async req(
    method: string,
    path: string,
    opts: { html?: string; title?: string } = {},
  ): Promise<Response> {
    const headers: Record<string, string> = {
      authorization: `Bearer ${this.token}`,
    };
    if (opts.html !== undefined) headers["content-type"] = "text/html";
    if (opts.title) headers["x-kiosk-title"] = opts.title;
    return fetch(`${this.apiBase}${path}`, { method, headers, body: opts.html });
  }

  private async parse(res: Response): Promise<unknown> {
    if (res.ok) return res.status === 204 ? undefined : res.json();
    let message = `HTTP ${res.status}`;
    try {
      const body = (await res.json()) as { error?: { message?: string } };
      message = body?.error?.message ?? message;
    } catch {
      // non-JSON error body
    }
    throw new CliError(message, res.status === 401 ? 2 : 1);
  }

  async publish(html: string, title?: string): Promise<PageResult> {
    return (await this.parse(
      await this.req("POST", "/pages", { html, title }),
    )) as PageResult;
  }
  async update(id: string, html: string, title?: string): Promise<PageResult> {
    return (await this.parse(
      await this.req("PUT", `/pages/${id}`, { html, title }),
    )) as PageResult;
  }
  async list(): Promise<{ pages: unknown[] }> {
    return (await this.parse(await this.req("GET", "/pages"))) as {
      pages: unknown[];
    };
  }
  async del(id: string): Promise<void> {
    await this.parse(await this.req("DELETE", `/pages/${id}`));
  }
  async me(): Promise<unknown> {
    return this.parse(await this.req("GET", "/me"));
  }
}
```

import { readFileSync } from "node:fs";
import { basename } from "node:path";
import type { KioskClient } from "./client";
import { findProjectRoot, resolveIdentity, writeMapping, mappingKey } from "./state";

export type PublishDeps = { client: KioskClient; cwd: string };
export type PublishArgs = { file: string; id?: string; forceNew?: boolean; title?: string };

/** Create or update a page from a file; record the mapping; return the URL. */
export async function publishCommand(
  deps: PublishDeps,
  args: PublishArgs,
): Promise<string> {
  const root = findProjectRoot(deps.cwd);
  const html = readFileSync(args.file, "utf8");
  const title = args.title ?? basename(args.file);
  const identity = resolveIdentity({
    root,
    file: args.file,
    id: args.id,
    forceNew: args.forceNew,
  });

  const result =
    identity.mode === "update"
      ? await deps.client.update(identity.id, html, title)
      : await deps.client.publish(html, title);

  writeMapping(root, mappingKey(root, args.file), { id: result.id, url: result.url });
  return result.url;
}
```

  logout                Remove the saved token
`;

function requireToken(): { apiBase: string; token: string } {
  const cfg = resolveConfig();
  if (!cfg.token) {
    throw new CliError("Not logged in. Run `kiosk login` or set KIOSK_TOKEN.", 2);
  }
  return { apiBase: cfg.apiBase, token: cfg.token };
}

export async function runCli(argv: string[]): Promise<number> {
  const [command, ...rest] = argv;
  try {
    switch (command) {
      case "login": {
        const { values } = parseArgs({ args: rest, options: { token: { type: "string" } }, allowPositionals: false });
        const token = values.token ?? process.env.KIOSK_TOKEN;
        if (!token) throw new CliError("Provide a token: kiosk login --token <t> (or set KIOSK_TOKEN)");
        const cfg = resolveConfig();
        await new KioskClient(cfg.apiBase, token).me(); // validate
        saveConfig({ token, apiBase: cfg.apiBase });
        process.stdout.write("Logged in.\n");
        return 0;
      }
      case "logout": {
        saveConfig({ apiBase: resolveConfig().apiBase });
        process.stdout.write("Logged out.\n");
        return 0;
      }
      case "publish": {
        const { values, positionals } = parseArgs({
          args: rest,
          allowPositionals: true,
          options: {
            id: { type: "string" }, new: { type: "boolean" },
            title: { type: "string" }, json: { type: "boolean" },
          },
        });
        const file = positionals[0];
        if (!file) throw new CliError("Usage: kiosk publish <file>");
        const { apiBase, token } = requireToken();
        const url = await publishCommand(
          { client: new KioskClient(apiBase, token), cwd: process.cwd() },
          { file, id: values.id, forceNew: values.new, title: values.title },
        );
        process.stdout.write(values.json ? JSON.stringify({ url }) + "\n" : url + "\n");
        return 0;
      }
      case "list": {
        const { apiBase, token } = requireToken();
        const { pages } = await new KioskClient(apiBase, token).list();
        process.stdout.write(JSON.stringify(pages, null, 2) + "\n");
        return 0;
      }
      case "delete": {
        const { positionals } = parseArgs({ args: rest, allowPositionals: true });
        const id = positionals[0];
        if (!id) throw new CliError("Usage: kiosk delete <id>");
        const { apiBase, token } = requireToken();
        await new KioskClient(apiBase, token).del(id);
        process.stdout.write("Deleted.\n");
        return 0;
      }
      case "whoami": {
        const { apiBase, token } = requireToken();
        process.stdout.write(JSON.stringify(await new KioskClient(apiBase, token).me(), null, 2) + "\n");
        return 0;
      }
      default:
        process.stdout.write(USAGE);
        return command ? 1 : 0;
    }
  } catch (e) {
    const err = e as CliError;
    process.stderr.write(`kiosk: ${err.message}\n`);
    return typeof err.exitCode === "number" ? err.exitCode : 1;
  }
}
