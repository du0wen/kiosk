# Kiosk P1 · Plan 4b — CLI Browser Login (loopback authorization-code) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make `kiosk login` open the browser, log in via the web app (OAuth), and finish over a localhost callback — like `gh`/`claude`/`mcp login` — while keeping `--token`/`KIOSK_TOKEN` as the headless fallback.

**Architecture:** A loopback authorization-code flow. The CLI starts a `127.0.0.1` server + `state`, opens `APP_ORIGIN/cli/authorize`; that page (behind a Better Auth session) mints a **one-time code** and 302s back to the loopback; the CLI exchanges the code at `POST /api/cli/exchange` for a freshly minted API token. `redirect_uri` is restricted to loopback so the code can never leave the user's machine.

**Tech Stack:** Next.js server component + server action + route handler, Drizzle/Neon, Node `http`/`child_process` in the CLI, Vitest.

> Plan **4b of P1** (inserted after Plan 4). Consumes Plan 1 (`getSession`, `member`, `createToken`), Plan 4 (`@kiosk/cli` config). Additive: does **not** remove `--token`/`KIOSK_TOKEN`.

## Global Constraints

- **Additive, not a replacement:** `kiosk login` (no `--token`) → browser flow; `kiosk login --token <t>` and `KIOSK_TOKEN` still work for headless/agent/CI.
- `redirect_uri` accepted by `/cli/authorize` **must be loopback** (`127.0.0.1`/`localhost`/`::1`, http). Reject otherwise.
- One-time code: SHA-256-hashed at rest, short TTL (2 min), single-use (`used_at`).
- `state` generated by the CLI, echoed through authorize→callback, verified by the CLI.
- Exchange endpoint is **token-less** (protected by the one-time code) and lives on `APP_ORIGIN` under `/api/cli/*` (middleware already 404s it on `SERVE_ORIGIN`).
- Commits carry **no AI attribution**.
- **Manually verified (needs OAuth creds + a browser):** the end-to-end browser flow (like Task 8). Automated tests cover the code service, the exchange endpoint, and the CLI's pure pieces.

```typescript
import { and, eq, gt, isNull } from "drizzle-orm";
import { createHash, randomBytes } from "node:crypto";
import { db } from "@/db";
import { cliAuthCode, member } from "@/db/schema";
import { uuidv7 } from "@/lib/id";
import { createToken } from "@/lib/token";

const CODE_TTL_MS = 2 * 60 * 1000;

export class CliAuthError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "CliAuthError";
  }
}

export function isLoopback(redirectUri: string): boolean {
  try {
    const u = new URL(redirectUri);
    const host = u.hostname;
    return (
      (u.protocol === "http:" || u.protocol === "https:") &&
      (host === "127.0.0.1" || host === "localhost" || host === "::1")
    );
  } catch {
    return false;
  }
}

function hash(code: string): string {
  return createHash("sha256").update(code).digest("hex");
}

export async function createCliAuthCode(params: {
  userId: string;
  redirectUri: string;
}): Promise<string> {
  if (!isLoopback(params.redirectUri)) {
    throw new CliAuthError("redirect_uri must be a loopback address");
  }
  const [m] = await db
    .select()
    .from(member)
    .where(eq(member.userId, params.userId))
    .limit(1);
  if (!m) throw new CliAuthError("no organization for user");
  const code = randomBytes(32).toString("base64url");
  await db.insert(cliAuthCode).values({
    codeHash: hash(code),
    userId: params.userId,
    organizationId: m.organizationId,
    redirectUri: params.redirectUri,
    expiresAt: new Date(Date.now() + CODE_TTL_MS),
  });
  return code;
}

export async function exchangeCliAuthCode(
  code: string,
  deviceName = "CLI",
): Promise<{ token: string }> {
  const [row] = await db
    .select()
    .from(cliAuthCode)
    .where(
      and(
        eq(cliAuthCode.codeHash, hash(code)),
        isNull(cliAuthCode.usedAt),
        gt(cliAuthCode.expiresAt, new Date()),
      ),
    )
    .limit(1);
  if (!row) throw new CliAuthError("invalid or expired code");
  await db.update(cliAuthCode).set({ usedAt: new Date() }).where(eq(cliAuthCode.id, row.id));
  const { plaintext } = await createToken({
    userId: row.userId,
    organizationId: row.organizationId,
    name: deviceName,
  });
  return { token: plaintext };
}
```

import { exchangeCliAuthCode, CliAuthError } from "@/lib/cli-auth";

export async function POST(req: Request): Promise<Response> {
  try {
    const body = (await req.json().catch(() => ({}))) as {
      code?: string;
      device?: string;
    };
    if (!body.code) {
      return Response.json(
        { error: { code: "bad_request", message: "Missing code" } },
        { status: 400 },
      );
    }
    const { token } = await exchangeCliAuthCode(body.code, body.device);
    return Response.json({ token });
  } catch (e) {
    if (e instanceof CliAuthError) {
      return Response.json(
        { error: { code: "invalid_code", message: e.message } },
        { status: 400 },
      );
    }
    console.error("cli/exchange error", e);
    return Response.json(
      { error: { code: "internal", message: "Internal error" } },
      { status: 500 },
    );
  }
}

export default async function CliAuthorizePage({
  searchParams,
}: {
  searchParams: Promise<{ redirect_uri?: string; state?: string }>;
}) {
  const { redirect_uri, state } = await searchParams;
  const session = await getSession();
  if (!session) {
    const back = `/cli/authorize?redirect_uri=${encodeURIComponent(redirect_uri ?? "")}&state=${encodeURIComponent(state ?? "")}`;

    "use server";
    const s = await getSession();
    if (!s) throw new Error("Not authenticated");
    if (!redirect_uri) throw new Error("Missing redirect_uri");
    const code = await createCliAuthCode({ userId: s.user.id, redirectUri: redirect_uri });
    redirect(`${redirect_uri}?code=${encodeURIComponent(code)}&state=${encodeURIComponent(state ?? "")}`);
  }

  return (
    <main style={{ maxWidth: 380, margin: "10vh auto", display: "grid", gap: 16 }}>
      <h1>Authorize the Kiosk CLI</h1>
      <p>Signed in as {session!.user.email}. Approve to log in the CLI on this device.</p>
      <form action={authorize}>
        <button type="submit">Authorize</button>
      </form>
    </main>
  );
}

```typescript
import { createServer } from "node:http";
import { randomBytes } from "node:crypto";
import { spawn } from "node:child_process";
import { AddressInfo } from "node:net";

export function appOriginFrom(apiBase: string): string {
  return apiBase.replace(/\/api\/v1\/?$/, "");
}

export function buildAuthorizeUrl(
  appOrigin: string,
  redirectUri: string,
  state: string,
): string {
  const u = new URL(`${appOrigin}/cli/authorize`);
  u.searchParams.set("redirect_uri", redirectUri);
  u.searchParams.set("state", state);
  return u.toString();
}

export function parseCallback(
  reqUrl: string,
  expectedState: string,
): { code: string } {
  const u = new URL(reqUrl, "http://127.0.0.1");
  if (u.searchParams.get("state") !== expectedState) {
    throw new Error("state mismatch");
  }
  const code = u.searchParams.get("code");
  if (!code) throw new Error("missing code");
  return { code };
}

function openBrowser(url: string): void {
  const cmd =
    process.platform === "darwin"
      ? "open"
      : process.platform === "win32"
        ? "start"
        : "xdg-open";
  try {
    spawn(cmd, [url], { stdio: "ignore", detached: true }).unref();
  } catch {
    // browser could not be opened automatically; the URL is printed for the user
  }
}

/** Loopback browser login. Resolves to the minted token. */
export async function browserLogin(apiBase: string): Promise<string> {
  const appOrigin = appOriginFrom(apiBase);
  const state = randomBytes(16).toString("hex");

  return new Promise<string>((resolve, reject) => {
    const server = createServer((req, res) => {
      const url = req.url ?? "/";
      if (!url.startsWith("/callback")) {
        res.writeHead(404).end();
        return;
      }
      res.writeHead(200, { "content-type": "text/html" });
      res.end("<h1>Kiosk</h1><p>Login complete — you can close this tab.</p>");
      server.close();
      (async () => {
        const { code } = parseCallback(url, state);
        const r = await fetch(`${appOrigin}/api/cli/exchange`, {
          method: "POST",
          headers: { "content-type": "application/json" },
          body: JSON.stringify({ code, device: `CLI on ${process.env.HOSTNAME ?? "device"}` }),
        });
        if (!r.ok) throw new Error(`exchange failed: HTTP ${r.status}`);
        const { token } = (await r.json()) as { token: string };
        resolve(token);
      })().catch(reject);
    });

    server.on("error", reject);
    server.listen(0, "127.0.0.1", () => {
      const { port } = server.address() as AddressInfo;
      const redirectUri = `http://127.0.0.1:${port}/callback`;
      const authUrl = buildAuthorizeUrl(appOrigin, redirectUri, state);
      process.stdout.write(`Opening your browser to log in:\n  ${authUrl}\n`);
      openBrowser(authUrl);
    });
  });
}
```
